/* ---------- V2.0: API pro mobilní stránku na vlastní adrese (GitHub Pages) ----------
   Stránka posílá POST s JSON { fn, arg } a dostane JSON { ok, data } nebo { ok:false, error }.
   Volat jde jen funkce níže – nic jiného z tohoto projektu ani z hlavní aplikace. */
function mApi_() {
  return { mNames: mNames, mLogin: mLogin, mLogout: mLogout, mMyEntries: mMyEntries, mSubmit: mSubmit, mDelete: mDelete,
           mVehInit: mVehInit, mVehCheck: mVehCheck, mVehFuel: mVehFuel, mVehDelete: mVehDelete };
}
function doPost(e) {
  let res;
  try {
    const body = JSON.parse(e && e.postData && e.postData.contents || '{}');
    const fn = String(body.fn || '');
    const api = mApi_();
    if (!Object.prototype.hasOwnProperty.call(api, fn)) throw new Error('Neznámá akce.');
    res = { ok: true, data: api[fn](body.arg || {}) };
  } catch (err) {
    res = { ok: false, error: String(err && err.message || err) };
  }
  return ContentService.createTextOutput(JSON.stringify(res)).setMimeType(ContentService.MimeType.JSON);
}

/* ---------- pomocné ---------- */
function mSs_() { return SpreadsheetApp.openById(M_CFG.DOCHAZKA); }
function mNorm_(v) {
  return String(v || '').toLowerCase().normalize('NFD').replace(/[̀-ͯ]/g, '').replace(/[^a-z0-9]/g, '');
}
function mToday_() { return Utilities.formatDate(new Date(), M_CFG.TZ, 'yyyy-MM-dd'); }
function mIsoShift_(iso, days) {
  const p = iso.split('-').map(Number);
  const d = new Date(p[0], p[1] - 1, p[2] + days, 12);
  return Utilities.formatDate(d, M_CFG.TZ, 'yyyy-MM-dd');
}
function mTime_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, M_CFG.TZ, 'HH:mm');
  const t = String(v == null ? '' : v).trim().replace(/\s+/g, '');
  let m = t.match(/^(\d{1,2})[:.,](\d{1,2})$/);
  if (m) {
    const min = m[2].length === 1 ? Number(m[2]) * 10 : Number(m[2]);
    if (Number(m[1]) > 23 || min > 59) return '';
    return String(Number(m[1])).padStart(2, '0') + ':' + String(min).padStart(2, '0');
  }
  m = t.match(/^(\d{1,2})(\d{2})$/);
  if (m && Number(m[1]) <= 23 && Number(m[2]) <= 59) return m[1].padStart(2, '0') + ':' + m[2];
  m = t.match(/^(\d{1,2})$/);
  if (m && Number(m[1]) <= 23) return String(Number(m[1])).padStart(2, '0') + ':00';
  return '';
}
function mHours_(from, to, pause) {
  const a = from.split(':').map(Number), b = to.split(':').map(Number);
  let d = (b[0] * 60 + b[1]) - (a[0] * 60 + a[1]);
  if (d <= 0) d += 24 * 60;
  return Math.max(0, Math.round((d / 60 - (Number(pause) || 0)) * 100) / 100);
}

function mSubSheet_() {
  const ss = mSs_();
  let sh = ss.getSheetByName(M_SUB_SHEET_);
  if (!sh) {
    sh = ss.insertSheet(M_SUB_SHEET_);
    sh.getRange(1, 1, 1, M_HEAD_.length).setValues([M_HEAD_]);
    sh.setFrozenRows(1);
    sh.getRange('E:G').setNumberFormat('@');
    sh.setColumnWidth(4, 200); sh.setColumnWidth(10, 260); sh.setColumnWidth(14, 220);
  }
  return sh;
}
function mPinSheet_() {
  const ss = mSs_();
  let sh = ss.getSheetByName(M_PIN_SHEET_);
  if (!sh) {
    sh = ss.insertSheet(M_PIN_SHEET_);
    sh.getRange(1, 1, 1, 3).setValues([['Zaměstnanec', 'PIN', 'Změněno']]);
    sh.setFrozenRows(1);
    sh.getRange('B:B').setNumberFormat('@');
  }
  return sh;
}

/** Zaměstnanci Dopravy (list Zaměstnanci): jméno, pobočka; bez ukončených k dnešku. */
function mEmployees_() {
  const sh = mSs_().getSheetByName('Zaměstnanci');
  if (!sh || sh.getLastRow() < 2) return [];
  const today = mToday_();
  return sh.getRange(2, 1, sh.getLastRow() - 1, 8).getDisplayValues()
    .map(r => ({ name: String(r[0] || '').trim(), branch: String(r[2] || '').trim(), category: String(r[3] || '').trim(), end: String(r[7] || '').trim() }))
    .filter(e => e.name && (mNorm_(e.category) === 'doprava' || !e.category))
    .filter(e => {
      if (!e.end) return true;
      const m = e.end.match(/^(\d{1,2})\.\s*(\d{1,2})\.\s*(\d{4})$/) || e.end.match(/^(\d{4})-(\d{2})-(\d{2})$/);
      if (!m) return true;
      const iso = m[1].length === 4 ? `${m[1]}-${m[2]}-${m[3]}` : `${m[3]}-${String(m[2]).padStart(2, '0')}-${String(m[1]).padStart(2, '0')}`;
      return iso >= mIsoShift_(today, -M_CFG.DAYS_BACK);
    });
}
function mPins_() {
  const sh = mPinSheet_();
  const out = {};
  if (sh.getLastRow() < 2) return out;
  sh.getRange(2, 1, sh.getLastRow() - 1, 2).getDisplayValues().forEach(r => {
    const pin = String(r[1] || '').trim();
    if (r[0] && pin) out[mNorm_(r[0])] = pin;
  });
  return out;
}
function mSession_(token) {
  const name = token ? CacheService.getScriptCache().get('mtok:' + token) : null;
  if (!name) throw new Error('Přihlášení vypršelo – přihlaste se prosím znovu.');
  const emp = mEmployees_().find(e => mNorm_(e.name) === mNorm_(name));
  if (!emp) throw new Error('Zaměstnanec už není v seznamu – obraťte se na dispečink.');
  return emp;
}

/** Jména k přihlášení – jen zaměstnanci Dopravy, kteří mají nastavený PIN. */
function mNames() {
  const pins = mPins_();
  return mEmployees_().filter(e => pins[mNorm_(e.name)]).map(e => e.name)
    .sort((a, b) => a.localeCompare(b, 'cs'));
}

function mLogin(p) {
  const name = String(p && p.name || '').trim();
  const pin = String(p && p.pin || '').trim();
  if (!name || !pin) throw new Error('Vyberte jméno a zadejte PIN.');
  const cache = CacheService.getScriptCache();
  const lockKey = 'mlock:' + mNorm_(name), triesKey = 'mtries:' + mNorm_(name);
  if (cache.get(lockKey)) throw new Error('Příliš mnoho špatných pokusů. Zkuste to za ' + M_CFG.LOCK_MINUTES + ' minut nebo kontaktujte dispečink.');
  const emp = mEmployees_().find(e => mNorm_(e.name) === mNorm_(name));
  const good = emp && mPins_()[mNorm_(name)] === pin;
  if (!good) {
    const n = Number(cache.get(triesKey) || 0) + 1;
    cache.put(triesKey, String(n), M_CFG.LOCK_MINUTES * 60);
    if (n >= M_CFG.MAX_TRIES) { cache.put(lockKey, '1', M_CFG.LOCK_MINUTES * 60); cache.remove(triesKey); }
    throw new Error('Špatné jméno nebo PIN.');
  }
  cache.remove(triesKey);
  const token = Utilities.getUuid().replace(/-/g, '') + Utilities.getUuid().replace(/-/g, '');
  cache.put('mtok:' + token, emp.name, M_CFG.SESSION_HOURS * 3600);
  return { token: token, name: emp.name, branch: emp.branch, today: mToday_(), daysBack: M_CFG.DAYS_BACK };
}

function mLogout(p) {
  if (p && p.token) CacheService.getScriptCache().remove('mtok:' + p.token);
  return true;
}

/** Vlastní zápisy za posledních 31 dní (+ dnešek). */
function mMyEntries(p) {
  const emp = mSession_(p && p.token);
  const sh = mSubSheet_();
  const from = mIsoShift_(mToday_(), -M_CFG.DAYS_BACK - 7);
  const out = [];
  if (sh.getLastRow() >= 2) {
    sh.getRange(2, 1, sh.getLastRow() - 1, 14).getValues().forEach(r => {
      if (mNorm_(r[3]) !== mNorm_(emp.name)) return;
      const iso = r[4] instanceof Date ? Utilities.formatDate(r[4], M_CFG.TZ, 'yyyy-MM-dd') : String(r[4] || '').trim();
      if (iso < from) return;
      out.push({
        id: String(r[0]), sent: r[1] instanceof Date ? vSentTxt_(r[1]) : String(r[1] || ''),
        date: iso, from: mTime_(r[5]), to: mTime_(r[6]), pause: Number(r[7]) || 0, hours: Number(r[8]) || 0,
        note: String(r[9] || ''), state: String(r[10] || 'Ke schválení'), reason: String(r[13] || '')
      });
    });
  }
  out.sort((a, b) => b.date.localeCompare(a.date) || b.sent.localeCompare(a.sent));
  return { name: emp.name, today: mToday_(), entries: out };
}

/** Uloží zápis dne. Čekající zápis téhož dne se přepíše, převzatý/zamítnutý zůstane a přibude nový (oprava). */
function mSubmit(p) {
  const emp = mSession_(p && p.token);
  const iso = String(p.date || '').slice(0, 10);
  const today = mToday_();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(iso)) throw new Error('Vyberte den.');
  if (iso > today) throw new Error('Docházku nelze zapsat dopředu.');
  if (iso < mIsoShift_(today, -M_CFG.DAYS_BACK)) throw new Error('Tak starý den už zapsat nelze – obraťte se na dispečink.');
  const from = mTime_(p.from), to = mTime_(p.to);
  if (!from || !to) throw new Error('Vyplňte čas OD a DO (např. 7:30).');
  const pause = Math.max(0, Number(String(p.pause || '0').replace(',', '.')) || 0);
  if (pause > 4) throw new Error('Pauza se zadává v hodinách (např. 0,5).');
  const hours = mHours_(from, to, pause);
  if (!(hours > 0)) throw new Error('Z časů vychází 0 hodin – zkontrolujte OD, DO a pauzu.');
  if (hours > 18) throw new Error('Z časů vychází víc než 18 hodin – zkontrolujte OD a DO.');
  const note = String(p.note || '').trim().slice(0, 300);

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) throw new Error('Server je zaneprázdněný, zkuste to prosím znovu.');
  try {
    const sh = mSubSheet_();
    let row = 0;
    if (sh.getLastRow() >= 2) {
      const vals = sh.getRange(2, 1, sh.getLastRow() - 1, 11).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        const d = vals[i][4] instanceof Date ? Utilities.formatDate(vals[i][4], M_CFG.TZ, 'yyyy-MM-dd') : String(vals[i][4] || '').trim();
        if (d === iso && mNorm_(vals[i][3]) === mNorm_(emp.name) && String(vals[i][10] || 'Ke schválení') === 'Ke schválení') { row = i + 2; break; }
      }
    }
    const id = row ? String(sh.getRange(row, 1).getValue()) : Utilities.getUuid().slice(0, 8);
    const vals = [[id, new Date(), emp.branch || '', emp.name, iso, from, to, pause, hours, note, 'Ke schválení', '', '', '']];
    if (!row) {
      row = sh.getLastRow() + 1;
      sh.getRange(row, 5, 1, 3).setNumberFormat('@');
    }
    sh.getRange(row, 1, 1, 14).setValues(vals);
    SpreadsheetApp.flush();
  } finally {
    lock.releaseLock();
  }
  return mMyEntries(p);
}

/** Smaže vlastní zápis, dokud ho dispečer nepřevzal. */
function mDelete(p) {
  const emp = mSession_(p && p.token);
  const id = String(p && p.id || '');
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) throw new Error('Server je zaneprázdněný, zkuste to prosím znovu.');
  try {
    const sh = mSubSheet_();
    if (sh.getLastRow() >= 2) {
      const vals = sh.getRange(2, 1, sh.getLastRow() - 1, 11).getValues();
      for (let i = vals.length - 1; i >= 0; i--) {
        if (String(vals[i][0]) === id && mNorm_(vals[i][3]) === mNorm_(emp.name)) {
          if (String(vals[i][10] || 'Ke schválení') !== 'Ke schválení') throw new Error('Zápis už dispečink zpracoval – nelze ho smazat.');
          sh.deleteRow(i + 2);
          break;
        }
      }
    }
  } finally {
    lock.releaseLock();
  }
  return mMyEntries(p);
}

/** Text závady, jak se uloží do tabulky – např. „Provozní kontrola: Palivo – nízký stav“. */
function vDefectLabel_(g, item, m) { m = m || g.mark || 'závada'; return g.g + ': ' + item + (m === 'závada' ? '' : ' – ' + m); }
/* V3.8: marks = víc druhů označení (výbava: chybí / poškozeno); popis je povinný u „závada“ a „poškozeno“ */
function vMarks_(g) { return g.marks && g.marks.length ? g.marks : [g.mark || 'závada']; }
function vNeedNote_(m) { return m === 'závada' || m === 'poškozeno'; }
const V_PAYS_ = ['Tankovací karta MOL', 'Soukromá karta', 'Hotově'];
const V_OWN_PAY_ = /hotov|soukrom/i;   // řidič platil ze svého → fotka účtenky povinná (proplácí se)
const V_CONFIRM_ = 'Potvrzuji, že jsem vozidlo zkontroloval a uvedené údaje jsou pravdivé.';

function vSs_() {
  if (!V_CFG.DATA_ID) throw new Error('Vozidla zatím nejsou zapnutá – obraťte se na dispečink.');
  return SpreadsheetApp.openById(V_CFG.DATA_ID);
}
function vSheet_(key) {
  const ss = vSs_();
  let sh = ss.getSheetByName(V_SH_[key]);
  if (!sh) {
    sh = ss.insertSheet(V_SH_[key]);
    sh.getRange(1, 1, 1, V_HEAD_[key].length).setValues([V_HEAD_[key]]).setFontWeight('bold').setBackground('#eef4ff');
    sh.setFrozenRows(1);
    if (key === 'VEH') {
      sh.getRange(2, 1, V_VEH_SEED_.length, 6).setValues(V_VEH_SEED_.map(r => r.concat(['', 'ANO'])));
      sh.setColumnWidth(5, 380);
    }
    if (key === 'FUEL') sh.getRange('G:G').setNumberFormat('@');
  } else if (sh.getLastColumn() < V_HEAD_[key].length) {   // V3.17: doplnit nové sloupce do hlavičky
    const n = sh.getLastColumn();
    sh.getRange(1, n + 1, 1, V_HEAD_[key].length - n).setValues([V_HEAD_[key].slice(n)]).setFontWeight('bold').setBackground('#eef4ff');
  }
  return sh;
}
function vRows_(key, maxRows) {
  const sh = vSheet_(key);
  const last = sh.getLastRow();
  if (last < 2) return [];
  const n = Math.min(last - 1, maxRows || 100000);
  const start = last - n + 1;
  return sh.getRange(start, 1, n, V_HEAD_[key].length).getValues().map((r, i) => ({ row: start + i, v: r }));
}
function vVehicles_() {
  return vRows_('VEH').map(x => ({
    code: String(x.v[0] || '').trim(), name: String(x.v[1] || '').trim(), branch: String(x.v[2] || '').trim(),
    spz: String(x.v[3] || '').trim(), active: mNorm_(x.v[5] || 'ano') !== 'ne'
  })).filter(v => v.code && v.active);
}
function vSetting_(key, value) {
  const sh = vSheet_('SET');
  const rows = sh.getLastRow() >= 2 ? sh.getRange(2, 1, sh.getLastRow() - 1, 2).getValues() : [];
  const i = rows.findIndex(r => String(r[0]) === key);
  if (value === undefined) return i >= 0 ? String(rows[i][1] || '') : '';
  if (i >= 0) sh.getRange(i + 2, 2).setValue(value); else sh.appendRow([key, value]);
  return value;
}
function vPhotoFolder_() {
  const props = PropertiesService.getScriptProperties();
  let id = props.getProperty('V_PHOTO_FOLDER') || vSetting_('Složka fotek (ID)');
  if (id) { try { const f = DriveApp.getFolderById(id); if (!f.isTrashed()) return f; } catch (e) {} }
  const file = DriveApp.getFileById(V_CFG.DATA_ID);
  const parents = file.getParents();
  const parent = parents.hasNext() ? parents.next() : DriveApp.getRootFolder();
  const f = parent.createFolder('Vozidla – fotky');
  props.setProperty('V_PHOTO_FOLDER', f.getId());
  vSetting_('Složka fotek (ID)', f.getId());
  return f;
}
function vSavePhotos_(photos, prefix, max) {
  const list = Array.isArray(photos) ? photos.slice(0, max) : [];
  if (!list.length) return [];
  const folder = vPhotoFolder_();
  return list.map((ph, i) => {
    const b64 = String(ph && ph.b64 || '').replace(/^data:[^,]*,/, '');
    if (!b64) return '';
    const bytes = Utilities.base64Decode(b64);
    if (bytes.length > V_CFG.MAX_PHOTO_MB * 1024 * 1024) throw new Error('Fotka je příliš velká.');
    const blob = Utilities.newBlob(bytes, 'image/jpeg', prefix + (list.length > 1 ? '_' + (i + 1) : '') + '.jpg');
    return folder.createFile(blob).getId();
  }).filter(Boolean);
}
function vKm_(v) {
  const n = Number(String(v == null ? '' : v).replace(/\s/g, '').replace(',', '.'));
  return Number.isFinite(n) ? Math.round(n) : NaN;
}
function vIso_(v) {
  if (v instanceof Date) return Utilities.formatDate(v, M_CFG.TZ, 'yyyy-MM-dd');
  const t = String(v || '').trim();
  let m = t.match(/^(\d{4})-(\d{2})-(\d{2})/); if (m) return m[0].slice(0, 10);
  m = t.match(/^(\d{1,2})\.\s*(\d{1,2})\.\s*(\d{4})/);
  return m ? `${m[3]}-${String(m[2]).padStart(2, '0')}-${String(m[1]).padStart(2, '0')}` : '';
}
function vSentTxt_(v) {   // „25.10.2026 v 17:51 hod.“ / „ve 23:45 hod.“
  if (!(v instanceof Date)) return String(v || '');
  const h = Number(Utilities.formatDate(v, M_CFG.TZ, 'H'));
  const pre = [2, 3, 4, 12, 13, 14, 20, 21, 22, 23].indexOf(h) >= 0 ? ' ve ' : ' v ';
  return Utilities.formatDate(v, M_CFG.TZ, 'd.M.yyyy') + pre + Utilities.formatDate(v, M_CFG.TZ, 'H:mm') + ' hod.';
}
function vSafe_(s) { return mNorm_(s).slice(0, 20) || 'x'; }
function vEmpBranchMatch_(empBranch, vehBranch) {
  const a = mNorm_(empBranch), b = mNorm_(vehBranch);
  return !a || !b || a.indexOf(b) >= 0 || b.indexOf(a) >= 0;
}

/** Úvod vozidel v mobilu: vozidla pobočky, poslední známé km, kontrolní seznam a moje poslední záznamy. */
function mVehInit(p) {
  const emp = mSession_(p && p.token);
  const all = vVehicles_();
  // V3.2: všechna vozidla (převozy mezi pobočkami) – nejdřív vozy vlastní pobočky, pak ostatní
  const own = all.filter(v => vEmpBranchMatch_(emp.branch, v.branch));
  const veh = own.concat(all.filter(v => own.indexOf(v) < 0));
  const lastKm = {};
  const chk = vRows_('CHK', 3000), fuel = vRows_('FUEL', 3000);
  chk.forEach(x => { const k = vKm_(x.v[6]); if (k > (lastKm[x.v[3]] || 0)) lastKm[x.v[3]] = k; });
  fuel.forEach(x => { if (String(x.v[12]) === 'Zamítnuto') return; const k = vKm_(x.v[7]); if (k > (lastKm[x.v[3]] || 0)) lastKm[x.v[3]] = k; });
  return {
    vehicles: veh.map(v => ({ code: v.code, name: v.name, spz: v.spz, branch: v.branch, lastKm: lastKm[v.code] || 0 })),
    groups: V_GROUPS_, pays: V_PAYS_, confirm: V_CONFIRM_,
    today: mToday_(), daysBack: M_CFG.DAYS_BACK,
    mine: vMine_(emp, chk, fuel),
    known: vKnown_(chk)
  };
}
/** V3.13: otevřené závady po vozidlech – { M3: { nz: true/false, items: { 'Položka (skupina: položka)': {date, who, state, note} } } }.
    Klíč je „Skupina: položka“ (bez „– chybí“ apod.), aby šel najít v kontrolním seznamu. */
function vKnown_(chk) {
  const out = {};
  chk.forEach(x => {
    const state = String(x.v[12] || '');
    if (state !== 'Nová závada' && state !== 'Řeší se') return;
    const code = String(x.v[3] || ''); if (!code) return;
    const o = out[code] = out[code] || { nz: false, items: {} };
    if (String(x.v[11] || '').toUpperCase() === 'NE') o.nz = true;
    const date = x.v[1] instanceof Date ? Utilities.formatDate(x.v[1], M_CFG.TZ, 'd.M.yyyy') : '';
    const done = {};   // V3.17: položky, které garážmistr už vyřešil
    String(x.v[16] || '').split('\n').forEach(l => { const k = l.split(' | ')[0].trim(); if (k) done[k] = 1; });
    const notes = {};
    String(x.v[9] || '').split('\n').forEach(l => { const m = l.match(/^([^:(]+?)(?:\s*\([^)]*\))?:\s*(.+)$/); if (m) notes[m[1].trim()] = m[2].trim(); });
    String(x.v[7] || '').split('\n').filter(Boolean).forEach(d => {
      if (done[d.trim()]) return;
      const key = d.replace(/\s+–\s+[^–]+$/, '').trim();        // bez „– chybí“, „– poškozeno“…
      const mark = (d.match(/\s+–\s+([^–]+)$/) || [])[1] || 'závada';
      const item = key.replace(/^[^:]+:\s*/, '');
      o.items[key] = { date: date, who: String(x.v[5] || '').split(/\s+/)[0], state: state, mark: mark, note: notes[item] || '' };
    });
  });
  return out;
}
function vMine_(emp, chk, fuel) {
  const from = mIsoShift_(mToday_(), -M_CFG.DAYS_BACK);
  const me = mNorm_(emp.name);
  const out = [];
  (chk || vRows_('CHK', 3000)).forEach(x => {
    if (mNorm_(x.v[5]) !== me) return;
    const iso = vIso_(x.v[1]); if (iso < from) return;
    out.push({ kind: 'check', id: String(x.v[0]), sentIso: x.v[1] instanceof Date ? x.v[1].toISOString() : '', sent: vSentTxt_(x.v[1]),
      vehicle: String(x.v[3]), km: vKm_(x.v[6]), defects: Number(x.v[8]) || 0, ok: String(x.v[11] || ''), state: String(x.v[12] || '') });
  });
  (fuel || vRows_('FUEL', 3000)).forEach(x => {
    if (mNorm_(x.v[5]) !== me) return;
    const iso = vIso_(x.v[6]); if (iso < from) return;
    out.push({ kind: 'fuel', id: String(x.v[0]), sentIso: x.v[1] instanceof Date ? x.v[1].toISOString() : '', sent: vSentTxt_(x.v[1]),
      vehicle: String(x.v[3]), date: iso, km: vKm_(x.v[7]), liters: Number(x.v[8]) || 0, pay: String(x.v[9] || ''),
      receipt: !!String(x.v[10] || ''), state: String(x.v[12] || 'Ke schválení'), reason: String(x.v[16] || '') });
  });
  out.sort((a, b) => String(b.sentIso).localeCompare(String(a.sentIso)));
  return out.slice(0, 30);
}
function vVehicleFor_(emp, code) {
  const v = vVehicles_().find(x => x.code === String(code || ''));
  if (!v) throw new Error('Vyberte vozidlo.');
  return v;
}
function vIdExists_(key, id) {
  return vRows_(key, 400).some(x => String(x.v[0]) === id);
}

/** Kontrola vozidla. p: {token, uid, vehicle, km, defects:["Skupina: položka"], desc, ok:'ANO'|'NE', confirm:true, photos:[{b64}]} */
function mVehCheck(p) {
  const emp = mSession_(p && p.token);
  const veh = vVehicleFor_(emp, p.vehicle);
  const km = vKm_(p.km);
  if (!(km > 0 && km < 3000000)) throw new Error('Zadejte stav tachometru (km).');
  const allowed = {}, grpOf = {};   // přijme i tvar bez „– nízký stav“ (starší stránka v mezipaměti mobilu)
  V_GROUPS_.forEach(g => g.items.forEach(i => {
    const first = vDefectLabel_(g, i, vMarks_(g)[0]);
    allowed[g.g + ': ' + i] = first;
    vMarks_(g).forEach(m => { const l = vDefectLabel_(g, i, m); allowed[l] = l; grpOf[l] = { g: g, item: i, mark: m }; });
  }));
  const rawNotes = p.notes && typeof p.notes === 'object' ? p.notes : null;
  const notes = {};
  if (rawNotes) Object.keys(rawNotes).forEach(k => { if (allowed[k]) notes[allowed[k]] = String(rawNotes[k] || '').trim().slice(0, 400); });
  const defects = (Array.isArray(p.defects) ? p.defects : []).map(String).filter(d => allowed[d]).map(d => allowed[d])
    .filter((d, i, a) => a.indexOf(d) === i)
    .filter((d, i, a) => !a.some((x, j) => j < i && grpOf[x].g === grpOf[d].g && grpOf[x].item === grpOf[d].item));   // jedna položka jen jednou
  const general = String(p.desc || '').trim().slice(0, 1500);
  let ok = String(p.ok || '').toUpperCase();
  if (defects.length) {
    if (rawNotes) {        // V3.7: popis u každé závady (u „chybí“ a „nízký stav“ nepovinný)
      const miss = defects.filter(d => vNeedNote_(grpOf[d].mark) && (notes[d] || '').length < 3).map(d => grpOf[d].item);
      if (miss.length) throw new Error('Popište závadu: ' + miss.join(', ') + '.');
    } else if (general.length < 3) throw new Error('Popište prosím zjištěnou závadu.');   // starší stránka
    if (ok !== 'ANO' && ok !== 'NE') throw new Error('Vyberte, zda je vozidlo technicky způsobilé k provozu.');
  } else ok = ok === 'NE' ? 'NE' : 'ANO';
  if (ok === 'NE' && !defects.length && general.length < 3) throw new Error('Popište prosím, proč vozidlo není způsobilé k provozu.');
  // do sloupce „Popis závady“: každá položka s popisem na vlastním řádku, pak doplňující popis
  const lines = defects.filter(d => notes[d]).map(d => grpOf[d].item + (grpOf[d].mark === 'závada' ? '' : ' (' + grpOf[d].mark + ')') + ': ' + notes[d]);
  if (general) lines.push((lines.length ? 'Poznámka: ' : '') + general);
  const desc = lines.join('\n').slice(0, 3000);
  if (p.confirm !== true) throw new Error('Potvrďte provedení kontroly.');
  const uid = String(p.uid || '').replace(/[^a-z0-9]/gi, '').slice(0, 20) || Utilities.getUuid().slice(0, 8);

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) throw new Error('Server je zaneprázdněný, zkuste to prosím znovu.');
  try {
    if (!vIdExists_('CHK', uid)) {
      const prefix = mToday_() + '_' + veh.code + '_kontrola_' + vSafe_(emp.name);
      // V3.11: obecné fotky (max. 4) + fotky u jednotlivých závad (max. 2 na položku, celkem max. 12)
      const photos = vSavePhotos_(p.photos, prefix, 4);
      const byItem = {};
      (Array.isArray(p.defPhotos) ? p.defPhotos : []).forEach(x => {
        const l = allowed[String(x && x.label || '')];
        if (!l || defects.indexOf(l) < 0) return;
        (byItem[l] = byItem[l] || []).push(x);
      });
      Object.keys(byItem).forEach(l => {
        if (photos.length >= 12) return;
        const ids = vSavePhotos_(byItem[l].slice(0, Math.min(2, 12 - photos.length)), prefix + '_' + vSafe_(grpOf[l].item), 2);
        photos.push.apply(photos, ids.map(id => id + '|' + l));   // V3.17: fotka zná svou závadu
      });
      const state = (defects.length || ok === 'NE') ? 'Nová závada' : 'Bez závad';
      vSheet_('CHK').appendRow([uid, new Date(), veh.branch || emp.branch || '', veh.code, veh.spz, emp.name, km,
        defects.join('\n'), defects.length, desc, photos.join(','), ok, state, '', '', '', '']);
      SpreadsheetApp.flush();
    }
  } finally { lock.releaseLock(); }
  return mVehInit(p);
}

/** Tankování. p: {token, uid, vehicle, date, km, liters, pay, note, photo:{b64}} */
function mVehFuel(p) {
  const emp = mSession_(p && p.token);
  const veh = vVehicleFor_(emp, p.vehicle);
  const iso = String(p.date || '').slice(0, 10), today = mToday_();
  if (!/^\d{4}-\d{2}-\d{2}$/.test(iso)) throw new Error('Vyberte datum tankování.');
  if (iso > today) throw new Error('Tankování nelze zapsat dopředu.');
  if (iso < mIsoShift_(today, -M_CFG.DAYS_BACK)) throw new Error('Tak staré tankování už zapsat nelze – obraťte se na dispečink.');
  const km = vKm_(p.km);
  if (!(km > 0 && km < 3000000)) throw new Error('Zadejte stav tachometru (km).');
  const liters = Math.round(Number(String(p.liters || '').replace(',', '.')) * 100) / 100;
  if (!(liters > 0 && liters <= 400)) throw new Error('Zadejte natankované litry (např. 72,5).');
  const pay = V_PAYS_.indexOf(String(p.pay || '')) >= 0 ? String(p.pay) : '';
  if (!pay) throw new Error('Vyberte způsob platby.');
  if (V_OWN_PAY_.test(pay) && !(p.photo && p.photo.b64)) throw new Error('Při platbě hotově nebo soukromou kartou vyfoťte prosím účtenku.');
  const note = String(p.note || '').trim().slice(0, 300);
  const uid = String(p.uid || '').replace(/[^a-z0-9]/gi, '').slice(0, 20) || Utilities.getUuid().slice(0, 8);

  const lock = LockService.getScriptLock();
  if (!lock.tryLock(25000)) throw new Error('Server je zaneprázdněný, zkuste to prosím znovu.');
  try {
    if (!vIdExists_('FUEL', uid)) {
      const prefix = iso + '_' + veh.code + '_uctenka_' + vSafe_(emp.name);
      const photos = vSavePhotos_(p.photo ? [p.photo] : [], prefix, 1);
      const sh = vSheet_('FUEL');
      const row = sh.getLastRow() + 1;
      sh.getRange(row, 7).setNumberFormat('@');
      sh.getRange(row, 1, 1, V_HEAD_.FUEL.length).setValues([[uid, new Date(), veh.branch || emp.branch || '', veh.code, veh.spz, emp.name,
        iso, km, liters, pay, photos.join(','), note, 'Ke schválení', '', '', '', '']]);
      SpreadsheetApp.flush();
    }
  } finally { lock.releaseLock(); }
  return mVehInit(p);
}

/** Smaže vlastní tankování, dokud ho dispečer nezapsal do karty. */
function mVehDelete(p) {
  const emp = mSession_(p && p.token);
  const id = String(p && p.id || '');
  const lock = LockService.getScriptLock();
  if (!lock.tryLock(20000)) throw new Error('Server je zaneprázdněný, zkuste to prosím znovu.');
  try {
    const x = vRows_('FUEL', 3000).reverse().find(r => String(r.v[0]) === id && mNorm_(r.v[5]) === mNorm_(emp.name));
    if (x) {
      if (String(x.v[12] || 'Ke schválení') !== 'Ke schválení') throw new Error('Tankování už dispečink zpracoval – nelze ho smazat.');
      String(x.v[10] || '').split(',').filter(Boolean).forEach(fid => { try { DriveApp.getFileById(fid).setTrashed(true); } catch (e) {} });
      vSheet_('FUEL').deleteRow(x.row);
    }
  } finally { lock.releaseLock(); }
  return mVehInit(p);
}
